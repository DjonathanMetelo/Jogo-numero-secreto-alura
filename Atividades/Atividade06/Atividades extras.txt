    Criar uma função que recebe três números como parâmetros e retorna a média deles.

    R:
    numeroRecebido1 = 5;
    numeroRecebido2 = 9;
    numeroRecebido3 = 7;
    triploNumero(numeroRecebido1, numeroRecebido2, numeroRecebido3);

    function triploNumero(dividendo1, dividendo2, dividendo3){
        let numero = 0;
        numero = (dividendo1 + dividendo2 + dividendo3)/3;
        return console.log(numero);
    };

    Criar uma função que recebe dois números como parâmetros e retorna o maior deles.

    R:

    numeroRecebido2 = 9;;
    maiorNumero(numeroRecebido1, numeroRecebido2);

    function maiorNumero(numero1, numero2){
        if (numero1 > numero2)
            return console.log(numero1);
        else
            return console.log(numero2);
    };
