    Crie uma função que calcule o valor do fatorial de um número passado como parâmetro.

    R:
    let fatorial = 1
    console.log(fatoriais(10));

    function fatoriais(numero){
        let numeroFatorial = numero
        do{
            if (numero == 1 || numero == 0){
                return `${numeroFatorial}! = ${fatorial}`;
            }
            fatorial = fatorial*numero;
            numero--
        } while(numero >= 1);   
    }

    Crie uma função que converte um valor em dólar, passado como parâmetro, e retorna o valor equivalente em reais. Para isso, considere a cotação do dólar igual a R$4,80.

    R:
    const cotacao = 4.80;
    console.log(conversorDolar(4.80));

    function conversorDolar(valor){
        let real = valor/cotacao;
        return parseFloat(real.toFixed(2));
    }

    Crie uma função que mostre na tela a área e o perímetro de uma sala retangular, utilizando altura e largura que serão dadas como parâmetro.

    R:
    console.log(areaEPerimetro(4,2));

    function areaEPerimetro(largura, altura){
        let area = largura*altura;
        let perimetro = 2*(largura+altura);
        return `Area: ${area}, Perimetro: ${perimetro}`;
    }

    Crie uma função que mostre na tela a área e o perímetro de uma sala circular, utilizando seu raio que será fornecido como parâmetro. Considere Pi = 3,14.

    R:
    const pi = 3.14;
    console.log(areaEPerimetro(5))

    function areaEPerimetro(raio){
        let area = pi*(raio*raio);
        let perimetro = 2*(pi*raio);
        return `Area: ${parseFloat(area.toFixed(2))}, Perimetro: ${parseFloat(perimetro.toFixed(2))}`;
    }
